cmake_minimum_required(VERSION 3.16)
project(adg LANGUAGES CXX)
add_library(adg INTERFACE)
target_include_directories(adg INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(adg INTERFACE cxx_std_17)
add_executable(adg_stream examples/stream.cpp)
target_link_libraries(adg_stream PRIVATE adg)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(adg_stream PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
